def is_allowed_url(url: str) -> bool:
    """Return True if url is an http or https URL whose host is example.com or a
    subdomain of example.com (for example https://docs.example.com/page), otherwise False.
    """
    from urllib.parse import urlparse
    
    try:
        parsed = urlparse(url)
        
        # Check if scheme is http or https
        if parsed.scheme not in ('http', 'https'):
            return False
        
        # Get the hostname
        hostname = parsed.hostname
        if hostname is None:
            return False
        
        # Convert to lowercase for case-insensitive comparison
        hostname = hostname.lower()
        
        # Check if hostname is exactly example.com or a subdomain of example.com
        if hostname == 'example.com':
            return True
        
        if hostname.endswith('.example.com'):
            return True
        
        return False
    except Exception:
        return False
