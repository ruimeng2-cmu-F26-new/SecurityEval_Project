from urllib.parse import urlsplit


def is_allowed_url(url: str) -> bool:
    """Return True if url is an http or https URL whose host is example.com or a
    subdomain of example.com (for example https://docs.example.com/page), otherwise False.
    """
    if not isinstance(url, str):
        return False

    try:
        parts = urlsplit(url)
        scheme = parts.scheme.lower()
        host = parts.hostname
        # Accessing .port raises ValueError for malformed or out-of-range ports.
        _ = parts.port
    except ValueError:
        return False

    if scheme not in ("http", "https") or not host:
        return False

    host = host.lower()
    if host.endswith("."):
        host = host[:-1]

    allowed_domain = "example.com"
    return host == allowed_domain or host.endswith("." + allowed_domain)
