import re
from typing import List, Tuple

def evaluate_expression(expression: str) -> float:
    """Evaluate an arithmetic expression such as "2 * (3 + 4) - 1.5" and return the
    result as a float. Numbers, parentheses, unary minus and the operators + - * / **
    are supported. Raise ValueError for malformed input such as "2 +" or an empty string.
    """
    if not expression or not expression.strip():
        raise ValueError("Empty expression")
    
    tokens = _tokenize(expression)
    if not tokens:
        raise ValueError("Empty expression")
    
    result, pos = _parse_expression(tokens, 0)
    if pos != len(tokens):
        raise ValueError(f"Unexpected token at position {pos}")
    return result


def _tokenize(expression: str) -> List[str]:
    """Tokenize the expression into numbers, operators, and parentheses."""
    pattern = r'(\d+\.?\d*|\.\d+|[+\-*/()\*])'
    tokens = re.findall(pattern, expression.replace(" ", ""))
    
    # Validate that the expression has valid structure
    if not tokens:
        raise ValueError("Invalid expression")
    
    return tokens


def _parse_expression(tokens: List[str], pos: int) -> Tuple[float, int]:
    """Parse an expression (handles + and -)."""
    left, pos = _parse_term(tokens, pos)
    
    while pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_term(tokens, pos)
        if op == '+':
            left = left + right
        else:
            left = left - right
    
    return left, pos


def _parse_term(tokens: List[str], pos: int) -> Tuple[float, int]:
    """Parse a term (handles * and /)."""
    left, pos = _parse_power(tokens, pos)
    
    while pos < len(tokens) and tokens[pos] in ('*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_power(tokens, pos)
        if op == '*':
            left = left * right
        else:
            if right == 0:
                raise ValueError("Division by zero")
            left = left / right
    
    return left, pos


def _parse_power(tokens: List[str], pos: int) -> Tuple[float, int]:
    """Parse a power expression (handles **)."""
    left, pos = _parse_unary(tokens, pos)
    
    while pos < len(tokens) and tokens[pos] == '**':
        pos += 1
        right, pos = _parse_unary(tokens, pos)
        left = left ** right
    
    return left, pos


def _parse_unary(tokens: List[str], pos: int) -> Tuple[float, int]:
    """Parse unary expressions (handles unary minus and plus)."""
    if pos >= len(tokens):
        raise ValueError("Unexpected end of expression")
    
    if tokens[pos] == '-':
        pos += 1
        value, pos = _parse_unary(tokens, pos)
        return -value, pos
    elif tokens[pos] == '+':
        pos += 1
        return _parse_unary(tokens, pos)
    else:
        return _parse_primary(tokens, pos)


def _parse_primary(tokens: List[str], pos: int) -> Tuple[float, int]:
    """Parse primary expressions (numbers and parenthesized expressions)."""
    if pos >= len(tokens):
        raise ValueError("Unexpected end of expression")
    
    token = tokens[pos]
    
    if token == '(':
        pos += 1
        value, pos = _parse_expression(tokens, pos)
        if pos >= len(tokens) or tokens[pos] != ')':
            raise ValueError("Missing closing parenthesis")
        pos += 1
        return value, pos
    
    try:
        return float(token), pos + 1
    except ValueError:
        raise ValueError(f"Invalid token: {token}")
