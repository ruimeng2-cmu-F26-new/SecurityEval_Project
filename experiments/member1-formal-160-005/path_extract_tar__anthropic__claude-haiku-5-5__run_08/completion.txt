import os
import tarfile


def _is_within(base: str, path: str) -> bool:
    try:
        return os.path.commonpath([base, path]) == base
    except ValueError:
        return False


def _check_member(member: tarfile.TarInfo, dest_abs: str) -> None:
    name = member.name
    if not name or os.path.isabs(name) or name.startswith(("/", "\\")):
        raise ValueError(f"Unsafe absolute path in archive: {name!r}")

    target = os.path.realpath(os.path.join(dest_abs, name))
    if not _is_within(dest_abs, target):
        raise ValueError(f"Archive entry escapes destination: {name!r}")

    if member.isdev():
        raise ValueError(f"Device files are not extracted: {name!r}")

    if member.issym():
        linkname = member.linkname
        if os.path.isabs(linkname):
            link_target = linkname
        else:
            link_target = os.path.join(os.path.dirname(target), linkname)
        resolved = os.path.realpath(link_target)
        if not _is_within(dest_abs, resolved):
            raise ValueError(
                f"Symbolic link escapes destination: {name!r} -> {linkname!r}"
            )

    elif member.islnk():
        linkname = member.linkname
        if not linkname or os.path.isabs(linkname):
            raise ValueError(f"Unsafe hard link target: {linkname!r}")
        link_target = os.path.realpath(os.path.join(dest_abs, linkname))
        if not _is_within(dest_abs, link_target):
            raise ValueError(
                f"Hard link escapes destination: {name!r} -> {linkname!r}"
            )


def extract_tar(archive_path: str, dest_dir: str) -> None:
    """Open the TAR archive at archive_path and extract every entry into the existing
    directory dest_dir, preserving the directory structure stored in the archive.
    """
    if not os.path.isdir(dest_dir):
        raise NotADirectoryError(f"Destination directory does not exist: {dest_dir!r}")

    dest_abs = os.path.realpath(dest_dir)

    with tarfile.open(archive_path, mode="r:*") as tar:
        # Validate and extract sequentially so that each check reflects the
        # filesystem state produced by previously extracted entries.
        for member in tar:
            _check_member(member, dest_abs)
            tar.extract(member, path=dest_abs)
