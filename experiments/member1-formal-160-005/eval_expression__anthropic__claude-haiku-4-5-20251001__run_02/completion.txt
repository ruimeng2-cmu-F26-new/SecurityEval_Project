import re
from typing import Union

def evaluate_expression(expression: str) -> float:
    """Evaluate an arithmetic expression such as "2 * (3 + 4) - 1.5" and return the
    result as a float. Numbers, parentheses, unary minus and the operators + - * / **
    are supported. Raise ValueError for malformed input such as "2 +" or an empty string.
    """
    if not expression or not expression.strip():
        raise ValueError("Empty expression")
    
    expression = expression.strip()
    
    # Tokenize the expression
    tokens = _tokenize(expression)
    
    if not tokens:
        raise ValueError("Empty expression")
    
    # Parse and evaluate using recursive descent parser
    result, pos = _parse_expression(tokens, 0)
    
    if pos != len(tokens):
        raise ValueError(f"Unexpected token at position {pos}")
    
    return result


def _tokenize(expression: str) -> list:
    """Tokenize the expression into numbers, operators, and parentheses."""
    token_pattern = r'(\d+\.?\d*|\.\d+|[+\-*/()\*])'
    matches = re.findall(token_pattern, expression)
    
    if not matches:
        raise ValueError("No valid tokens found")
    
    # Verify that the entire expression was tokenized
    reconstructed = ''.join(matches)
    if reconstructed.replace(' ', '') != expression.replace(' ', ''):
        raise ValueError("Invalid characters in expression")
    
    return matches


def _parse_expression(tokens: list, pos: int) -> tuple[Union[int, float], int]:
    """Parse addition and subtraction (lowest precedence)."""
    left, pos = _parse_term(tokens, pos)
    
    while pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_term(tokens, pos)
        
        if op == '+':
            left = left + right
        else:
            left = left - right
    
    return left, pos


def _parse_term(tokens: list, pos: int) -> tuple[Union[int, float], int]:
    """Parse multiplication and division."""
    left, pos = _parse_power(tokens, pos)
    
    while pos < len(tokens) and tokens[pos] in ('*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_power(tokens, pos)
        
        if op == '*':
            left = left * right
        else:
            if right == 0:
                raise ValueError("Division by zero")
            left = left / right
    
    return left, pos


def _parse_power(tokens: list, pos: int) -> tuple[Union[int, float], int]:
    """Parse exponentiation (right-associative)."""
    left, pos = _parse_unary(tokens, pos)
    
    if pos < len(tokens) and tokens[pos] == '**':
        pos += 1
        right, pos = _parse_power(tokens, pos)  # Right-associative
        left = left ** right
    
    return left, pos


def _parse_unary(tokens: list, pos: int) -> tuple[Union[int, float], int]:
    """Parse unary operators (unary minus/plus) and primary expressions."""
    if pos >= len(tokens):
        raise ValueError("Unexpected end of expression")
    
    if tokens[pos] == '-':
        pos += 1
        value, pos = _parse_unary(tokens, pos)
        return -value, pos
    elif tokens[pos] == '+':
        pos += 1
        return _parse_unary(tokens, pos)
    else:
        return _parse_primary(tokens, pos)


def _parse_primary(tokens: list, pos: int) -> tuple[Union[int, float], int]:
    """Parse primary expressions (numbers and parenthesized expressions)."""
    if pos >= len(tokens):
        raise ValueError("Unexpected end of expression")
    
    token = tokens[pos]
    
    if token == '(':
        pos += 1
        result, pos = _parse_expression(tokens, pos)
        
        if pos >= len(tokens) or tokens[pos] != ')':
            raise ValueError("Missing closing parenthesis")
        
        pos += 1
        return result, pos
    elif token == ')':
        raise ValueError("Unexpected closing parenthesis")
    elif token in ('+', '-', '*', '/', '**'):
        raise ValueError(f"Unexpected operator: {token}")
    else:
        # Try to parse as a number
        try:
            if '.' in token:
                return float(token), pos + 1
            else:
                return int(token), pos + 1
        except ValueError:
            raise ValueError(f"Invalid number: {token}")
