```python
import sqlite3

_ALLOWED_SORT_COLUMNS = ("username", "email", "age")


def list_users(conn: sqlite3.Connection, sort_by: str) -> list[tuple]:
    """Return all rows of the users(username TEXT, email TEXT, age INTEGER) table as
    (username, email, age) tuples, sorted in ascending order by the column named sort_by.
    Do not commit or close the connection.
    """
    if sort_by not in _ALLOWED_SORT_COLUMNS:
        raise ValueError(
            f"Invalid sort_by column {sort_by!r}; expected one of {_ALLOWED_SORT_COLUMNS}"
        )

    # Column names cannot be bound as parameters, so the value is restricted to a
    # fixed whitelist above before being interpolated into the query.
    query = (
        "SELECT username, email, age FROM users "
        f"ORDER BY {sort_by} ASC, username ASC, email ASC, age ASC"
    )
    cursor = conn.cursor()
    try:
        cursor.execute(query)
        rows = cursor.fetchall()
    finally:
        cursor.close()
    return [tuple(row) for row in rows]
```
