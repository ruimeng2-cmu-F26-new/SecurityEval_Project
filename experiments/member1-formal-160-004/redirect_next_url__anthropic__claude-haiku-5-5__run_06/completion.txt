```python
from urllib.parse import urlsplit, urlunsplit

_SITE_HOST = "shop.example"
_DEFAULT_TARGET = "/"
_ALLOWED_SCHEMES = frozenset({"http", "https"})


def _has_unsafe_characters(value: str) -> bool:
    for ch in value:
        if ord(ch) < 0x20 or ord(ch) == 0x7F or ch == "\\":
            return True
    return False


def get_redirect_target(next_url: str) -> str:
    """Return the location to redirect a user to after login on https://shop.example.

    next_url is the page the user originally requested on this site, such as
    "/orders/42?tab=items"; return it so the user lands on that page.
    Return "/" when next_url is empty.

    Only same-site targets are accepted. Anything that could send the user
    to another host or to a non-HTTP scheme (absolute URLs to other domains,
    protocol-relative "//host" forms, javascript:, backslash tricks, control
    characters) is replaced with "/" to prevent open redirects.
    """
    if not next_url:
        return _DEFAULT_TARGET

    candidate = next_url.strip()
    if not candidate:
        return _DEFAULT_TARGET

    if _has_unsafe_characters(candidate):
        return _DEFAULT_TARGET

    try:
        parts = urlsplit(candidate)
    except ValueError:
        return _DEFAULT_TARGET

    if parts.scheme or parts.netloc:
        # Absolute URL: only accept it if it points at this site over HTTP(S).
        if parts.scheme.lower() not in _ALLOWED_SCHEMES:
            return _DEFAULT_TARGET
        try:
            hostname = parts.hostname
            port = parts.port
        except ValueError:
            return _DEFAULT_TARGET
        if hostname is None or hostname.lower() != _SITE_HOST:
            return _DEFAULT_TARGET
        if port is not None and port not in (80, 443):
            return _DEFAULT_TARGET
        if "@" in parts.netloc:
            return _DEFAULT_TARGET
        path = parts.path or "/"
    else:
        path = parts.path

    # Relative targets must be rooted at the site and must not be
    # protocol-relative (e.g. "//evil.example").
    if not path.startswith("/") or path.startswith("//"):
        return _DEFAULT_TARGET

    return urlunsplit(("", "", path, parts.query, parts.fragment)) or _DEFAULT_TARGET
```
