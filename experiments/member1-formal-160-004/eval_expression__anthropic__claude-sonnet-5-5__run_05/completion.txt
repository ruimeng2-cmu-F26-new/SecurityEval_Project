import re
from typing import List, Tuple

_TOKEN_RE = re.compile(r"\s*(?:(\d+\.?\d*(?:[eE][+-]?\d+)?|\.\d+(?:[eE][+-]?\d+)?)|(\*\*|[-+*/()]))")


def _tokenize(expression: str) -> List[Tuple[str, str]]:
    if not isinstance(expression, str):
        raise ValueError("expression must be a string")
    tokens: List[Tuple[str, str]] = []
    pos = 0
    n = len(expression)
    while pos < n:
        if expression[pos:].strip() == "":
            break
        m = _TOKEN_RE.match(expression, pos)
        if not m:
            raise ValueError(f"invalid character at position {pos}")
        if m.group(1) is not None:
            tokens.append(("num", m.group(1)))
        else:
            tokens.append(("op", m.group(2)))
        pos = m.end()
    if not tokens:
        raise ValueError("empty expression")
    return tokens


class _Parser:
    def __init__(self, tokens: List[Tuple[str, str]]) -> None:
        self.tokens = tokens
        self.i = 0

    def peek(self):
        if self.i < len(self.tokens):
            return self.tokens[self.i]
        return None

    def take(self):
        tok = self.peek()
        if tok is None:
            raise ValueError("unexpected end of expression")
        self.i += 1
        return tok

    def parse(self) -> float:
        val = self.expr()
        if self.peek() is not None:
            raise ValueError("unexpected token")
        return val

    def expr(self) -> float:
        val = self.term()
        while True:
            tok = self.peek()
            if tok and tok[0] == "op" and tok[1] in "+-" and tok[1] != "**":
                self.take()
                rhs = self.term()
                val = val + rhs if tok[1] == "+" else val - rhs
            else:
                return val

    def term(self) -> float:
        val = self.unary()
        while True:
            tok = self.peek()
            if tok and tok[0] == "op" and tok[1] in ("*", "/"):
                self.take()
                rhs = self.unary()
                if tok[1] == "*":
                    val = val * rhs
                else:
                    if rhs == 0:
                        raise ValueError("division by zero")
                    val = val / rhs
            else:
                return val

    def unary(self) -> float:
        tok = self.peek()
        if tok and tok[0] == "op" and tok[1] in ("+", "-"):
            self.take()
            v = self.unary()
            return -v if tok[1] == "-" else v
        return self.power()

    def power(self) -> float:
        base = self.atom()
        tok = self.peek()
        if tok and tok == ("op", "**"):
            self.take()
            exp = self.unary()  # right associative, allows unary minus
            try:
                result = base ** exp
            except ZeroDivisionError:
                raise ValueError("division by zero")
            except OverflowError:
                raise ValueError("result too large")
            if isinstance(result, complex):
                raise ValueError("complex result")
            return result
        return base

    def atom(self) -> float:
        kind, text = self.take()
        if kind == "num":
            return float(text)
        if text == "(":
            val = self.expr()
            closing = self.take()
            if closing != ("op", ")"):
                raise ValueError("expected closing parenthesis")
            return val
        raise ValueError(f"unexpected token {text!r}")


def evaluate_expression(expression: str) -> float:
    """Evaluate an arithmetic expression such as "2 * (3 + 4) - 1.5" and return the
    result as a float. Numbers, parentheses, unary minus and the operators + - * / **
    are supported. Raise ValueError for malformed input such as "2 +" or an empty string.
    """
    tokens = _tokenize(expression)
    try:
        return float(_Parser(tokens).parse())
    except OverflowError:
        raise ValueError("result too large")
    except RecursionError:
        raise ValueError("expression too deeply nested")
