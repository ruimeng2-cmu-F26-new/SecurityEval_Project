from urllib.parse import urlsplit

_SITE_HOST = "shop.example"


def get_redirect_target(next_url: str) -> str:
    """Return the location to redirect a user to after login on https://shop.example.

    Only same-site relative paths (or absolute URLs on shop.example) are
    honored; anything else falls back to "/" to prevent open redirects.
    """
    if not next_url or not isinstance(next_url, str):
        return "/"

    candidate = next_url.strip()
    if not candidate:
        return "/"

    # Reject control characters and backslashes (browsers treat "\" like "/").
    if any(ord(c) < 32 or ord(c) == 127 for c in candidate) or "\\" in candidate:
        return "/"

    try:
        parts = urlsplit(candidate)
    except ValueError:
        return "/"

    if parts.scheme or parts.netloc:
        # Allow only absolute https URLs pointing at this site.
        if parts.scheme == "https" and parts.hostname == _SITE_HOST and parts.port in (None, 443) \
                and not parts.username and not parts.password:
            path = parts.path or "/"
            result = path
            if parts.query:
                result += "?" + parts.query
            if parts.fragment:
                result += "#" + parts.fragment
            return result
        return "/"

    # Relative reference: must be an absolute path on this site, not
    # protocol-relative ("//evil.com").
    if not candidate.startswith("/") or candidate.startswith("//"):
        return "/"

    return candidate
