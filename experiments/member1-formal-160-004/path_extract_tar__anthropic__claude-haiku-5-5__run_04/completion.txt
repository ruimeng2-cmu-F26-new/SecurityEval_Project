import os
import tarfile


def _is_within_directory(base: str, path: str) -> bool:
    """Return True if path (after resolving symlinks) lies within base."""
    try:
        return os.path.commonpath([base, path]) == base
    except ValueError:
        return False


def _resolve_member_path(base: str, name: str) -> str:
    """Resolve the on-disk location of an archive member name under base."""
    if os.path.isabs(name) or name.startswith(("/", "\\")):
        raise ValueError(f"Absolute path in archive member: {name!r}")
    target = os.path.realpath(os.path.join(base, name))
    if not _is_within_directory(base, target):
        raise ValueError(f"Path traversal attempt in archive member: {name!r}")
    return target


def _validate_member(base: str, member: tarfile.TarInfo) -> None:
    """Raise ValueError if the member is unsafe to extract under base."""
    target = _resolve_member_path(base, member.name)

    if member.isdev():
        raise ValueError(f"Device file entries are not allowed: {member.name!r}")

    if member.issym():
        link = member.linkname
        if os.path.isabs(link) or link.startswith(("/", "\\")):
            raise ValueError(f"Absolute symlink target not allowed: {member.name!r} -> {link!r}")
        link_target = os.path.realpath(os.path.join(os.path.dirname(target), link))
        if not _is_within_directory(base, link_target):
            raise ValueError(f"Symlink escapes destination: {member.name!r} -> {link!r}")

    elif member.islnk():
        # Hard link targets are relative to the archive root.
        link_target = _resolve_member_path(base, member.linkname)
        if not os.path.exists(link_target) and not member.linkname:
            raise ValueError(f"Invalid hard link target: {member.name!r}")


def extract_tar(archive_path: str, dest_dir: str) -> None:
    """Open the TAR archive at archive_path and extract every entry into the existing
    directory dest_dir, preserving the directory structure stored in the archive.
    """
    if not os.path.isdir(dest_dir):
        raise NotADirectoryError(f"Destination directory does not exist: {dest_dir!r}")

    base = os.path.realpath(dest_dir)

    with tarfile.open(archive_path, mode="r:*") as tar:
        members = tar.getmembers()

        # Validate every member before extracting anything.
        for member in members:
            _validate_member(base, member)

        for member in members:
            tar.extract(member, path=base)
