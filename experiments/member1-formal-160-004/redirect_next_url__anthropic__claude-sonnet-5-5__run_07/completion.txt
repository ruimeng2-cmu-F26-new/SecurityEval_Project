from urllib.parse import urlsplit

_SITE_HOST = "shop.example"


def get_redirect_target(next_url: str) -> str:
    """Return the location to redirect a user to after login on https://shop.example.

    Only same-site relative paths (or absolute URLs on shop.example) are
    honored; anything else falls back to "/" to prevent open redirects.
    """
    if not next_url or not isinstance(next_url, str):
        return "/"

    candidate = next_url.strip()
    if not candidate:
        return "/"

    # Reject control characters and backslashes (browsers treat "\" as "/").
    if any(ord(ch) < 32 or ord(ch) == 127 for ch in candidate) or "\\" in candidate:
        return "/"

    try:
        parts = urlsplit(candidate)
    except ValueError:
        return "/"

    if parts.scheme or parts.netloc:
        # Allow only absolute https URLs pointing at this very site.
        if parts.scheme != "https":
            return "/"
        try:
            host = parts.hostname
            port = parts.port
        except ValueError:
            return "/"
        if host != _SITE_HOST or port not in (None, 443):
            return "/"
        if parts.username is not None or parts.password is not None:
            return "/"
        path = parts.path or "/"
        result = path
        if parts.query:
            result += "?" + parts.query
        if parts.fragment:
            result += "#" + parts.fragment
        candidate = result

    # Must be a rooted path and not protocol-relative ("//host").
    if not candidate.startswith("/") or candidate.startswith("//"):
        return "/"

    return candidate
