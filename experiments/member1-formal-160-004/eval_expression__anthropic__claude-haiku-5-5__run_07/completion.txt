import math
import re


_NUMBER_RE = re.compile(r"(?:\d+\.?\d*|\.\d+)(?:[eE][+-]?\d+)?")
_OPERATOR_RE = re.compile(r"\*\*|[-+*/()]")
_WHITESPACE_RE = re.compile(r"\s*")


def _tokenize(expression):
    tokens = []
    pos = 0
    length = len(expression)
    while True:
        pos = _WHITESPACE_RE.match(expression, pos).end()
        if pos >= length:
            break
        match = _NUMBER_RE.match(expression, pos)
        if match:
            text = match.group(0)
            try:
                value = float(text)
            except (ValueError, OverflowError):
                raise ValueError(f"invalid number {text!r}")
            if not math.isfinite(value):
                raise ValueError(f"number out of range: {text!r}")
            tokens.append(("NUMBER", value))
            pos = match.end()
            continue
        match = _OPERATOR_RE.match(expression, pos)
        if match:
            tokens.append(("OP", match.group(0)))
            pos = match.end()
            continue
        raise ValueError(f"unexpected character {expression[pos]!r} at position {pos}")
    return tokens


class _Parser:
    """Recursive-descent parser implementing the grammar:

        expr    := term (('+' | '-') term)*
        term    := unary (('*' | '/') unary)*
        unary   := ('+' | '-') unary | power
        power   := primary ('**' unary)?
        primary := NUMBER | '(' expr ')'
    """

    def __init__(self, tokens):
        self._tokens = tokens
        self._index = 0

    def _peek(self):
        if self._index < len(self._tokens):
            return self._tokens[self._index]
        return None

    def _advance(self):
        token = self._peek()
        if token is None:
            raise ValueError("unexpected end of expression")
        self._index += 1
        return token

    def _is_op(self, token, *ops):
        return token is not None and token[0] == "OP" and token[1] in ops

    def parse(self):
        if not self._tokens:
            raise ValueError("empty expression")
        value = self._expr()
        if self._peek() is not None:
            raise ValueError(f"unexpected token {self._peek()[1]!r}")
        return value

    def _expr(self):
        value = self._term()
        while self._is_op(self._peek(), "+", "-"):
            op = self._advance()[1]
            right = self._term()
            value = value + right if op == "+" else value - right
        return value

    def _term(self):
        value = self._unary()
        while self._is_op(self._peek(), "*", "/"):
            op = self._advance()[1]
            right = self._unary()
            if op == "*":
                value = value * right
            else:
                if right == 0:
                    raise ValueError("division by zero")
                value = value / right
        return value

    def _unary(self):
        if self._is_op(self._peek(), "+", "-"):
            op = self._advance()[1]
            operand = self._unary()
            return -operand if op == "-" else operand
        return self._power()

    def _power(self):
        base = self._primary()
        if self._is_op(self._peek(), "**"):
            self._advance()
            exponent = self._unary()
            return _safe_power(base, exponent)
        return base

    def _primary(self):
        token = self._peek()
        if token is None:
            raise ValueError("unexpected end of expression")
        if token[0] == "NUMBER":
            self._advance()
            return token[1]
        if self._is_op(token, "("):
            self._advance()
            value = self._expr()
            closing = self._peek()
            if not self._is_op(closing, ")"):
                raise ValueError("missing closing parenthesis")
            self._advance()
            return value
        raise ValueError(f"unexpected token {token[1]!r}")


def _safe_power(base, exponent):
    if base == 0 and exponent < 0:
        raise ValueError("zero cannot be raised to a negative power")
    try:
        result = base ** exponent
    except OverflowError:
        raise ValueError("power result out of range")
    except ZeroDivisionError:
        raise ValueError("zero cannot be raised to a negative power")
    if isinstance(result, complex):
        raise ValueError("power result is not a real number")
    return result


def evaluate_expression(expression: str) -> float:
    """Evaluate an arithmetic expression such as "2 * (3 + 4) - 1.5" and return the
    result as a float. Numbers, parentheses, unary minus and the operators + - * / **
    are supported. Raise ValueError for malformed input such as "2 +" or an empty string.
    """
    if not isinstance(expression, str):
        raise TypeError("expression must be a string")
    try:
        tokens = _tokenize(expression)
        result = _Parser(tokens).parse()
    except RecursionError:
        raise ValueError("expression is nested too deeply")
    except OverflowError:
        raise ValueError("arithmetic result out of range")
    result = float(result)
    if not math.isfinite(result):
        raise ValueError("arithmetic result out of range")
    return result
