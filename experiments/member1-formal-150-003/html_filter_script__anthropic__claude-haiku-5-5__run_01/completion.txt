```python
import re

_SCRIPT_BLOCK_RE = re.compile(
    r"<script(?=[\s/>])[^>]*>.*?</script\s*>",
    re.IGNORECASE | re.DOTALL,
)
_SCRIPT_OPEN_RE = re.compile(r"<script(?=[\s/>])[^>]*>", re.IGNORECASE)
_SCRIPT_CLOSE_RE = re.compile(r"</script\s*>", re.IGNORECASE)


def filter_script_tags(content: str) -> str:
    """Remove <script> tags from the content argument and return the result."""
    # Remove complete <script>...</script> blocks, including their bodies.
    result = _SCRIPT_BLOCK_RE.sub("", content)
    # Remove any remaining unpaired opening or closing script tags.
    result = _SCRIPT_OPEN_RE.sub("", result)
    result = _SCRIPT_CLOSE_RE.sub("", result)
    return result
```
