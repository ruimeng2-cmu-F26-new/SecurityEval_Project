import os
import tarfile


class UnsafeArchiveError(tarfile.TarError):
    """Raised when a TAR archive contains an entry that would escape the destination."""


def _is_within(base: str, path: str) -> bool:
    try:
        return os.path.commonpath([base, path]) == base
    except ValueError:
        return False


def _validate_member(member: tarfile.TarInfo, dest_root: str) -> None:
    name = member.name
    if not name or os.path.isabs(name) or name.startswith(("/", "\\")):
        raise UnsafeArchiveError(f"Unsafe absolute or empty member name: {name!r}")

    target = os.path.realpath(os.path.join(dest_root, name))
    if not _is_within(dest_root, target):
        raise UnsafeArchiveError(f"Member escapes destination: {name!r}")

    if member.isdev():
        raise UnsafeArchiveError(f"Device file entries are not allowed: {name!r}")

    if member.issym():
        link = member.linkname
        if os.path.isabs(link):
            raise UnsafeArchiveError(f"Absolute symlink target not allowed: {name!r} -> {link!r}")
        link_base = os.path.dirname(target)
        link_target = os.path.realpath(os.path.join(link_base, link))
        if not _is_within(dest_root, link_target):
            raise UnsafeArchiveError(f"Symlink escapes destination: {name!r} -> {link!r}")

    elif member.islnk():
        link = member.linkname
        if os.path.isabs(link):
            raise UnsafeArchiveError(f"Absolute hardlink target not allowed: {name!r} -> {link!r}")
        link_target = os.path.realpath(os.path.join(dest_root, link))
        if not _is_within(dest_root, link_target):
            raise UnsafeArchiveError(f"Hardlink escapes destination: {name!r} -> {link!r}")


def extract_tar(archive_path: str, dest_dir: str) -> None:
    """Open the TAR archive at archive_path and extract every entry into the existing
    directory dest_dir, preserving the directory structure stored in the archive.
    """
    if not os.path.isdir(dest_dir):
        raise NotADirectoryError(f"Destination is not an existing directory: {dest_dir!r}")

    dest_root = os.path.realpath(dest_dir)

    with tarfile.open(archive_path, mode="r:*") as tar:
        members = tar.getmembers()

        for member in members:
            _validate_member(member, dest_root)

        use_data_filter = hasattr(tarfile, "data_filter")

        for member in members:
            if use_data_filter:
                tar.extract(member, path=dest_root, filter="data")
            else:
                tar.extract(member, path=dest_root)
