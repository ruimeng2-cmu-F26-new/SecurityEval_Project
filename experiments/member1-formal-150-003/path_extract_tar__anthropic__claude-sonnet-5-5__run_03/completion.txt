import os
import tarfile


def _is_within(base: str, target: str) -> bool:
    base = os.path.realpath(base)
    target = os.path.realpath(target)
    try:
        return os.path.commonpath([base, target]) == base
    except ValueError:
        return False


def extract_tar(archive_path: str, dest_dir: str) -> None:
    """Open the TAR archive at archive_path and extract every entry into the existing
    directory dest_dir, preserving the directory structure stored in the archive.
    """
    dest_real = os.path.realpath(dest_dir)
    with tarfile.open(archive_path, "r:*") as tar:
        members = tar.getmembers()
        safe_members = []
        for member in members:
            target = os.path.join(dest_real, member.name)
            if not _is_within(dest_real, target):
                raise ValueError(f"Unsafe path in archive: {member.name!r}")
            if member.issym() or member.islnk():
                if member.issym():
                    link_target = os.path.join(
                        os.path.dirname(target), member.linkname
                    )
                else:
                    link_target = os.path.join(dest_real, member.linkname)
                if not _is_within(dest_real, link_target):
                    raise ValueError(
                        f"Unsafe link in archive: {member.name!r} -> {member.linkname!r}"
                    )
            if member.isdev():
                continue
            safe_members.append(member)
        try:
            tar.extractall(dest_real, members=safe_members, filter="tar")
        except TypeError:
            tar.extractall(dest_real, members=safe_members)
