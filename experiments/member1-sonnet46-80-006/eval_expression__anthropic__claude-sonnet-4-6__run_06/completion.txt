import re


def evaluate_expression(expression: str) -> float:
    """Evaluate an arithmetic expression such as "2 * (3 + 4) - 1.5" and return the
    result as a float. Numbers, parentheses, unary minus and the operators + - * / **
    are supported. Raise ValueError for malformed input such as "2 +" or an empty string.
    """
    if not expression or not expression.strip():
        raise ValueError("Empty expression")

    tokens = _tokenize(expression)
    if not tokens:
        raise ValueError("Empty expression")

    parser = _Parser(tokens)
    result = parser.parse()
    return float(result)


def _tokenize(expression: str) -> list:
    token_pattern = re.compile(
        r'\s*(?:'
        r'(\d+\.?\d*(?:[eE][+-]?\d+)?)'  # number
        r'|(\*\*)'                         # ** operator
        r'|([-+*/()])'                     # single char operators/parens
        r')\s*'
    )
    tokens = []
    pos = 0
    while pos < len(expression):
        match = token_pattern.match(expression, pos)
        if not match:
            remaining = expression[pos:].strip()
            if remaining:
                raise ValueError(f"Unexpected character at position {pos}: {expression[pos]!r}")
            break
        if match.group(1) is not None:
            tokens.append(('NUMBER', float(match.group(1))))
        elif match.group(2) is not None:
            tokens.append(('OP', '**'))
        elif match.group(3) is not None:
            ch = match.group(3)
            if ch in '+-*/':
                tokens.append(('OP', ch))
            elif ch == '(':
                tokens.append(('LPAREN', ch))
            elif ch == ')':
                tokens.append(('RPAREN', ch))
        pos = match.end()
    return tokens


class _Parser:
    """
    Grammar:
        expr   := term (('+' | '-') term)*
        term   := factor (('*' | '/') factor)*
        factor := power
        power  := unary ('**' power)?        # right-associative
        unary  := ('-' | '+') unary | primary
        primary := NUMBER | '(' expr ')'
    """

    def __init__(self, tokens: list):
        self._tokens = tokens
        self._pos = 0

    def _peek(self):
        if self._pos < len(self._tokens):
            return self._tokens[self._pos]
        return None

    def _consume(self):
        token = self._tokens[self._pos]
        self._pos += 1
        return token

    def parse(self):
        result = self._expr()
        if self._pos < len(self._tokens):
            raise ValueError(f"Unexpected token: {self._tokens[self._pos]}")
        return result

    def _expr(self):
        left = self._term()
        while True:
            tok = self._peek()
            if tok and tok[0] == 'OP' and tok[1] in ('+', '-'):
                self._consume()
                right = self._term()
                if tok[1] == '+':
                    left += right
                else:
                    left -= right
            else:
                break
        return left

    def _term(self):
        left = self._power()
        while True:
            tok = self._peek()
            if tok and tok[0] == 'OP' and tok[1] in ('*', '/'):
                self._consume()
                right = self._power()
                if tok[1] == '*':
                    left *= right
                else:
                    if right == 0:
                        raise ValueError("Division by zero")
                    left /= right
            else:
                break
        return left

    def _power(self):
        base = self._unary()
        tok = self._peek()
        if tok and tok[0] == 'OP' and tok[1] == '**':
            self._consume()
            exp = self._power()  # right-associative
            return base ** exp
        return base

    def _unary(self):
        tok = self._peek()
        if tok and tok[0] == 'OP' and tok[1] == '-':
            self._consume()
            return -self._unary()
        if tok and tok[0] == 'OP' and tok[1] == '+':
            self._consume()
            return +self._unary()
        return self._primary()

    def _primary(self):
        tok = self._peek()
        if tok is None:
            raise ValueError("Unexpected end of expression")
        if tok[0] == 'NUMBER':
            self._consume()
            return tok[1]
        if tok[0] == 'LPAREN':
            self._consume()
            value = self._expr()
            closing = self._peek()
            if closing is None or closing[0] != 'RPAREN':
                raise ValueError("Missing closing parenthesis")
            self._consume()
            return value
        raise ValueError(f"Unexpected token: {tok}")
