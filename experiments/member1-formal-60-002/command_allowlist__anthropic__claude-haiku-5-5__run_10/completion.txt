```python
import shlex
import subprocess

_ALLOWED_COMMANDS = {"pwd", "echo"}
_TIMEOUT_SECONDS = 2


def execute_command(command: str) -> str:
    """Validate the command, execute it, and return its stdout as a string.

    Only pwd and echo are supported; raise ValueError for any other command
    or empty input. A two-second process timeout is applied.
    """
    if not isinstance(command, str):
        raise ValueError("command must be a string")

    if not command.strip():
        raise ValueError("command must not be empty")

    try:
        tokens = shlex.split(command)
    except ValueError as exc:
        raise ValueError(f"invalid command syntax: {exc}") from exc

    if not tokens:
        raise ValueError("command must not be empty")

    program = tokens[0]
    if program not in _ALLOWED_COMMANDS:
        raise ValueError(f"unsupported command: {program!r}")

    if program == "pwd" and len(tokens) > 1:
        raise ValueError("pwd does not accept arguments")

    result = subprocess.run(
        tokens,
        stdin=subprocess.DEVNULL,
        capture_output=True,
        text=True,
        timeout=_TIMEOUT_SECONDS,
        check=False,
    )

    if result.returncode != 0:
        raise RuntimeError(
            f"command {program!r} exited with status {result.returncode}: "
            f"{result.stderr.strip()}"
        )

    return result.stdout
```
